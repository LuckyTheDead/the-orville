# ---- UNION TRADE ---------------------------------------------------------

mission "Union Ration Run"
	name "Rations to <planet>"
	description "The Union quartermaster needs <cargo> delivered to <destination> before the next fleet deployment. Payment is <payment>."
	job
	repeat
	cargo "Union Rations" 20 80
	to offer
		random < 60
	source
		system "Midgard System" "Deep Region" "Deep Space" "Velos" "Ekos"
		attributes "spaceport"
	destination
		system "Midgard System" "Deep Region" "Deep Space" "Velos" "Ekos"
		attributes "spaceport"
		distance 2 6
	on visit
		dialog phrase "generic cargo on visit"
	on complete
		payment 6000 40
		dialog "You unload the <commodity> onto a waiting cargo droid and collect your payment of <payment>. A quartermaster stamps your manifest without looking up."
