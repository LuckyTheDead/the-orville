event "union arc begins"
	date 1 1 3013
	set "union arc: start"

mission "Union The Home Front 0"
	landing
	non-blocking
	name "The Home Front"
	description "A Union colony needs medical supplies. Deliver them to <destination>."
	cargo "Union Mission Supplies" 5 20
	deadline
	source
		government "Republic"
		attributes "spaceport"
	to offer
		has "event: union arc begins"
	on offer
		conversation
			`A contact is waiting for you near the landing pads, and they are not the sort of person who waits twice.`
			`They have a job. They have the cargo already crated. What they do not have is a ship they trust.`
			choice
				`\t"I'll take it on."`
					accept
				`\t"What does the Union need?"`
					goto explain
				`\t"Not my business."`
					decline
			label explain
			`The colony's hospital was damaged in a storm and the nearest supply ship is three weeks out.`
				accept
	on complete
		payment 25000
		set "union arc: ch0"
		log "The Home Front: The colony's hospital was damaged in a storm and the nearest supply ship is three weeks out."
