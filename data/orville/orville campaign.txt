# The Orville: the main campaign.
#
# Twenty missions following the Union from the Krill frontier through
# first contact with the Kaylon, the Moclan crisis, the invasion and the
# coalition, to the armistice. Chained on the "orville arc: *" conditions.

event "orville arc begins"
	date 1 1 3013
	set "orville arc: start"

mission "Orville Border Fire 0"
	landing
	non-blocking
	name "The Raided Colony"
	description "A Krill raiding party has hit the Ekos colony. Deliver emergency supplies to <destination> and report what you find."
	cargo "Union Mission Supplies" 5 20
	deadline
	source
		government "Republic"
		attributes "spaceport"
	to offer
		has "event: orville arc begins"
	on offer
		conversation
			`A contact is waiting for you near the landing pads, and they are not the sort of person who waits twice.`
			`They have a job. They have the cargo already crated. What they do not have is a ship they trust.`
			choice
				`\t"I'll take the supplies up."`
					accept
				`\t"What happened out there?"`
					goto explain
				`\t"Not my department."`
					decline
			label explain
			`The colony's shield generator is still standing, but the Krill took the food stores.`
			`Three Krill cruisers came through the system at dawn. They did not hail anyone. They did not ask.`
				accept
	on complete
		payment 30000
		set "orville arc: ch0"
		log "Delivered emergency supplies to the Ekos colony after a Krill raid. The Krill are no longer raiding for food; they are raiding for position."
